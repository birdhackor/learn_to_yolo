"""Coordinator-controlled disclosure; shared filesystem is not technical isolation."""
import argparse
from datetime import datetime, timezone
import hashlib
import json
from pathlib import Path

RUN=Path(__file__).resolve().parent
ROOT=RUN.parents[2]
LOGS=ROOT/'reviews/clear-tutorial/16f6910/first-read'
LOGS.mkdir(parents=True,exist_ok=True)
p=argparse.ArgumentParser();p.add_argument('action',choices=['next','record','status']);p.add_argument('reader');p.add_argument('--file',type=Path)
a=p.parse_args()
packets=json.loads((RUN/f'{a.reader}-packets.json').read_text())
state_path=RUN/f'{a.reader}-state.json';state=json.loads(state_path.read_text())
if a.action=='status':
 print(json.dumps({'reader':a.reader,'completed':state['cursor'],'total':len(packets),'awaiting_record':state['awaiting_record']},ensure_ascii=False));raise SystemExit
if a.action=='next':
 if state['awaiting_record']:raise SystemExit('Record the current unit before opening another one.')
 if state['cursor']==len(packets):print('READING COMPLETE');raise SystemExit
 packet=packets[state['cursor']]; state['awaiting_record']=True
 state_path.write_text(json.dumps(state))
 print(json.dumps(packet,ensure_ascii=False,indent=2));raise SystemExit
if not state['awaiting_record']:raise SystemExit('No disclosed unit is awaiting a record.')
if a.file is None:raise SystemExit('--file is required')
note=json.loads(a.file.read_text())
packet=packets[state['cursor']]
assert note['unit_id']==packet['id'], 'Record does not match the disclosed unit'
required=['understanding','elements','next_change_and_basis','issues','needed_visuals','visual_status','variation']
assert all(key in note for key in required), 'Missing required first-read fields'
assert isinstance(note['issues'],list)
for issue in note['issues']:
 assert all(key in issue for key in ['quote','reason','severity','needed']), 'Issue must quote the current source and name the gap'
 assert issue['quote'] in packet['content'], 'Issue quote must be present in this disclosed unit'
 assert issue['severity'] in ['blocking','burden','optional']
metadata={key:value for key,value in packet.items() if key!='content'}
record={'disclosed_source':metadata,'recorded_at_utc':datetime.now(timezone.utc).isoformat(),'reader_notes':note}
with (LOGS/f'{a.reader}.jsonl').open('a') as stream:stream.write(json.dumps(record,ensure_ascii=False)+'\n')
state['cursor']+=1;state['awaiting_record']=False;state_path.write_text(json.dumps(state))
print(json.dumps({'saved':packet['id'],'completed':state['cursor'],'total':len(packets)},ensure_ascii=False))
