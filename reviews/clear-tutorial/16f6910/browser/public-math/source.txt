import asyncio
import json
from pathlib import Path
from playwright.async_api import async_playwright

OUT = Path('/workspace/learn_to_yolo/reviews/clear-tutorial/16f6910/browser/public-math')
OUT.mkdir(parents=True, exist_ok=True)
BASE = 'https://birdhackor.github.io/learn_to_yolo'
PAGES = ['03-identity', '04-coordinates', '07-loss', '11-iou-loss', '16-inference-head', '20-deployment']
SNAPSHOT = r'''() => {
 const article = document.querySelector('article');
 const chars = [...article.querySelectorAll('mjx-c')];
 const invisible = chars.filter(n => n.classList.contains('mjx-c2061') && getComputedStyle(n,'::before').content === '""');
 const missing = chars.filter(n => !invisible.includes(n) && ['none','normal','""'].includes(getComputedStyle(n,'::before').content));
 const items = Array.from(MathJax.startup.document.math);
 const wrappers = [...article.querySelectorAll('.arithmatex, .md-typeset__scrollwrap')].map(n => {
   const style = getComputedStyle(n);
   const b=n.getBoundingClientRect();
   const before=n.scrollLeft; n.scrollLeft=100000; const moved=n.scrollLeft; n.scrollLeft=before;
   return {tag:n.tagName, classes:n.className, text:n.textContent.slice(0,90),client:n.clientWidth,scroll:n.scrollWidth,overflow:style.overflowX,moved,left:b.left,right:b.right};
 });
 return {colabLinks:[...article.querySelectorAll('a[href*=\"colab.research.google.com/github/\"]')].map(a=>a.href),url:location.href,documentMarker:window.documentMarker,formulas:article.querySelectorAll('.arithmatex').length,
   containers:article.querySelectorAll('mjx-container').length,nested:article.querySelectorAll('mjx-container mjx-container').length,
   chars:chars.length,expectedInvisibleGlyphs:invisible.length,missingGlyphs:missing.map(n=>({classes:n.className,content:getComputedStyle(n,'::before').content})),
   mathItems:items.length,disconnectedItems:items.filter(i=>!i.typesetRoot?.isConnected).length,
   unrelatedItems:items.filter(i=>!article.contains(i.typesetRoot)).length,
   styleNodes:[...document.querySelectorAll('#MJX-CHTML-styles')].map(n=>({connected:n.isConnected,rules:n.sheet?.cssRules.length})),
   adaptiveCSS:MathJax.startup.output.options.adaptiveCSS,windowWidth:innerWidth,clientWidth:document.documentElement.clientWidth,documentWidth:document.documentElement.scrollWidth,
   tables:article.querySelectorAll('table').length,scrollable:wrappers.filter(n=>n.scroll>n.client+1),wrappers};
}'''

async def settled(page):
    slug = page.url.split('/lessons/', 1)[1].split('/', 1)[0]
    await page.wait_for_function("slug => !!document.querySelector('article a[href*=\"notebooks/' + slug + '.ipynb\"]')", arg=slug)
    await page.evaluate('() => new Promise(requestAnimationFrame)')
    await page.wait_for_function('window.MathJax && MathJax.startup && MathJax.startup.document')
    await page.evaluate('async () => { let p; do {p=MathJax.startup.promise; await p;} while(p!==MathJax.startup.promise); await document.fonts.ready; }')

async def snapshot(page, label):
    await settled(page)
    await page.locator('article details').evaluate_all('nodes => nodes.forEach(n => n.open=true)')
    result = await page.evaluate(SNAPSHOT)
    result['label'] = label
    return result

async def click_lesson(page, slug):
    link = page.locator(f'.md-nav--primary a[href$="/lessons/{slug}/"]').first
    await link.click(no_wait_after=True)
    await page.wait_for_url(f'**/lessons/{slug}/')
    await page.wait_for_function("slug => !!document.querySelector('article a[href*=\"notebooks/' + slug + '.ipynb\"]')", arg=slug)

async def main():
    result = {'desktop':[], 'mobile':[], 'console':[], 'pageErrors':[], 'documents':[]}
    async with async_playwright() as p:
        browser = await p.chromium.launch(executable_path='/usr/bin/chromium',args=['--no-sandbox','--disable-gpu','--disable-dev-shm-usage'])
        context = await browser.new_context(viewport={'width':1280,'height':900})
        page = await context.new_page()
        page.on('console',lambda m:result['console'].append({'type':m.type,'text':m.text,'url':page.url,'location':m.location}))
        page.on('pageerror',lambda e:result['pageErrors'].append(str(e)))
        page.on('request',lambda r:result['documents'].append(r.url) if r.resource_type=='document' else None)
        await page.goto(f'{BASE}/lessons/{PAGES[0]}/')
        await page.evaluate('window.documentMarker="published-v0.5.0-document"')
        result['desktop'].append(await snapshot(page,'initial'))
        for slug in PAGES[1:]:
            await click_lesson(page,slug)
            result['desktop'].append(await snapshot(page,f'click-{slug}'))
        await page.locator('article').screenshot(path=str(OUT/'20-desktop.png'),timeout=15000)
        # Click two sidebar links without waiting for MathJax between them.
        await click_lesson(page,'07-loss')
        await click_lesson(page,'11-iou-loss')
        result['desktop'].append(await snapshot(page,'rapid-two-clicks'))
        await page.go_back()
        await page.wait_for_url('**/lessons/07-loss/')
        result['desktop'].append(await snapshot(page,'browser-back'))
        anchor=page.locator('article a.headerlink').first
        await anchor.click(no_wait_after=True)
        await page.wait_for_function('location.hash.length>0')
        result['desktop'].append(await snapshot(page,'same-page-anchor'))
        # Observe subsequent native operations, then deliberately delay one job
        # so two queued articles can become disconnected before their turn.
        await page.evaluate(r'''() => {
          const orig=MathJax.typesetPromise.bind(MathJax), clear=MathJax.typesetClear.bind(MathJax);
          window.probe={events:[],active:0,max:0};
          MathJax.typesetClear=(...a)=>{probe.events.push({kind:'clear',before:Array.from(MathJax.startup.document.math).length});return clear(...a)};
          MathJax.typesetPromise=async a=>{probe.active++;probe.max=Math.max(probe.max,probe.active);probe.events.push({kind:'start',connected:a[0].isConnected,heading:a[0].querySelector('h1').textContent});await new Promise(r=>setTimeout(r,400));try{return await orig(a)}finally{probe.active--;probe.events.push({kind:'end',connected:a[0].isConnected})}};
        }''')
        await click_lesson(page,'04-coordinates')
        await page.wait_for_function('probe.active===1')
        await click_lesson(page,'11-iou-loss')
        await click_lesson(page,'16-inference-head')
        result['stressSnapshot']=await snapshot(page,'delayed-queue-stress')
        result['stressProbe']=await page.evaluate('probe')
        await context.close()
        mobile = await browser.new_context(viewport={'width':390,'height':844},device_scale_factor=1,is_mobile=True,has_touch=True)
        mp = await mobile.new_page()
        mp.on('console',lambda m:result['console'].append({'type':m.type,'text':m.text,'url':mp.url,'location':m.location,'mobile':True}))
        mp.on('pageerror',lambda e:result['pageErrors'].append(str(e)))
        for slug in PAGES:
            await mp.goto(f'{BASE}/lessons/{slug}/')
            result['mobile'].append(await snapshot(mp,f'mobile-{slug}'))
            if slug in ['07-loss','11-iou-loss','20-deployment']:
                await mp.locator('article').screenshot(path=str(OUT/f'{slug}-mobile.png'),timeout=15000)
        # Native horizontal wheel events demonstrate the overflow controls.
        await mp.goto(f'{BASE}/lessons/11-iou-loss/')
        await settled(mp)
        await mp.locator('article details').evaluate_all('nodes => nodes.forEach(n => n.open=true)')
        result['nativeScroll']=[]
        for kind,selector in [('formula','article div.arithmatex'),('table','article .md-typeset__scrollwrap')]:
            nodes=mp.locator(selector)
            index=await nodes.evaluate_all('ns=>ns.findIndex(n=>n.scrollWidth>n.clientWidth+50)')
            node=nodes.nth(index)
            await node.scroll_into_view_if_needed()
            await node.screenshot(path=str(OUT/f'{kind}-mobile-before.png'),timeout=15000)
            before=await node.evaluate('n=>n.scrollLeft')
            box=await node.bounding_box()
            await mp.mouse.move(box['x']+box['width']/2,max(100,min(794,box['y']+box['height']/2)))
            await mp.mouse.wheel(220,0)
            await mp.wait_for_timeout(250)
            after=await node.evaluate('n=>n.scrollLeft')
            await node.screenshot(path=str(OUT/f'{kind}-mobile-after.png'),timeout=15000)
            result['nativeScroll'].append({'kind':kind,'before':before,'after':after,'client':await node.evaluate('n=>n.clientWidth'),'scroll':await node.evaluate('n=>n.scrollWidth')})
        await browser.close()
    (OUT/'browser.json').write_text(json.dumps(result,ensure_ascii=False,indent=2)+'\n')
    print(json.dumps({k:[{x:y for x,y in s.items() if x not in ['wrappers','missingGlyphs','scrollable']}|{'missingGlyphs':len(s['missingGlyphs']),'scrollable':len(s['scrollable'])} for s in result[k]] for k in ['desktop','mobile']},ensure_ascii=False,indent=2))
    print(json.dumps({k:result[k] for k in ['console','pageErrors','documents','stressProbe','stressSnapshot']},ensure_ascii=False,indent=2))

asyncio.run(main())
