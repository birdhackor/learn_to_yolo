import json,hashlib
from pathlib import Path
from playwright.sync_api import sync_playwright
root=Path('/workspace/learn_to_yolo');out=root/'reviews/clear-tutorial/16f6910/browser';results=[]
with sync_playwright() as p:
 b=p.chromium.launch(executable_path='/usr/bin/chromium',args=['--no-sandbox','--disable-gpu','--disable-dev-shm-usage'])
 for width,height in [(1280,900),(390,844)]:
  page=b.new_page(viewport={'width':width,'height':height});errors=[];page.on('pageerror',lambda e:errors.append(str(e)))
  response=page.goto('http://127.0.0.1:8767/learn_to_yolo/lessons/07-heldout/');page.wait_for_selector('article');page.evaluate('async()=>{await MathJax.startup.promise;await document.fonts.ready}')
  body=page.locator('article').inner_text();assert '#k 紅／藍 score s' in body and '#1 紅 score 0.980' in body,body
  page.locator('article details').evaluate_all('ns=>ns.forEach(n=>n.open=true)')
  for image in page.locator('article img').all():image.evaluate('async n=>await n.decode()')
  r=page.evaluate('''()=>({url:location.href,width:innerWidth,documentWidth:document.documentElement.scrollWidth,formulas:document.querySelectorAll('article .arithmatex').length,containers:document.querySelectorAll('article mjx-container').length,nested:document.querySelectorAll('article mjx-container mjx-container').length,mathErrors:document.querySelectorAll('article mjx-merror').length,images:[...document.querySelectorAll('article img')].map(n=>({src:n.src,width:n.getBoundingClientRect().width,loaded:n.complete&&n.naturalWidth>0}))})''')
  assert r['documentWidth']<=width+1 and r['formulas']==r['containers'] and not r['nested'] and not r['mathErrors'] and not errors
  r.update(http=response.status,page_sha256=hashlib.sha256((root/'docs/lessons/07-heldout.md').read_bytes()).hexdigest(),red_blue_caption_present=True,image_zero_red_caption_present=True,pageErrors=errors);results.append(r)
  # Context captured at first revised caption; actual page UI stays visible.
  target=page.locator('article p').filter(has_text='#k 紅／藍 score s').first
  target.evaluate('n=>window.scrollTo(0,n.getBoundingClientRect().top+window.scrollY-100)');page.screenshot(path=str(out/f'final-heldout-{width}.png'))
  page.close()
 b.close()
(out/'final-heldout.json').write_text(json.dumps(results,ensure_ascii=False,indent=2)+'\n')
print(json.dumps({'checks':len(results),'passed':True}))
