from pathlib import Path
import json
from playwright.sync_api import sync_playwright
ROOT=Path('/workspace/learn_to_yolo');OUT=ROOT/'artifacts/runs/clear-tutorial-16f6910/whole-site-browser';OUT.mkdir(exist_ok=True)
BASE='http://127.0.0.1:8767/learn_to_yolo/'
pages=sorted(ROOT.joinpath('docs').rglob('*.md'))
results=[];errors=[]
with sync_playwright() as p:
 b=p.chromium.launch(executable_path='/usr/bin/chromium',args=['--no-sandbox','--disable-gpu','--disable-dev-shm-usage'])
 for width,height in [(1280,900),(390,844)]:
  page=b.new_page(viewport={'width':width,'height':height})
  page.on('pageerror',lambda e:errors.append(str(e)))
  for source in pages:
   route=source.relative_to(ROOT/'docs').with_suffix('').as_posix();route='' if route=='index' else route+'/'
   response=page.goto(BASE+route)
   page.wait_for_selector('article')
   if page.locator('article .arithmatex').count():
    page.wait_for_function('!!window.MathJax?.startup?.document')
    page.evaluate('async()=>{await MathJax.startup.promise;await document.fonts.ready}')
   else:page.evaluate('async()=>{await document.fonts.ready}')
   page.locator('article details').evaluate_all('nodes=>nodes.forEach(n=>n.open=true)')
   page.evaluate('async()=>{await Promise.all([...document.querySelectorAll("article img")].map(i=>i.decode().catch(()=>{})))}')
   result=page.evaluate('''()=>({url:location.href,documentWidth:document.documentElement.scrollWidth,windowWidth:innerWidth,formulas:document.querySelectorAll('article .arithmatex').length,math:document.querySelectorAll('article mjx-container').length,nested:document.querySelectorAll('article mjx-container mjx-container').length,mathErrors:document.querySelectorAll('article mjx-merror').length,images:[...document.querySelectorAll('article img')].map(n=>({src:n.src,width:n.getBoundingClientRect().width,height:n.getBoundingClientRect().height,naturalWidth:n.naturalWidth,loaded:n.complete&&n.naturalWidth>0}))})''')
   result.update(page=source.relative_to(ROOT).as_posix(),viewport=width,http=response.status)
   assert result['http']==200 and result['documentWidth']<=width+1,result
   assert result['formulas']==result['math'] and not result['nested'] and not result['mathErrors'],result
   assert all(i['loaded'] for i in result['images']),result
   # Capture every SVG figure in its actual page layout once per viewport.
   for i,img in enumerate(page.locator('article img').all()):
    name=img.get_attribute('src').split('/')[-1].split('?')[0]
    shot=OUT/f'{width}-{name}.png'
    if not shot.exists():img.screenshot(path=str(shot))
   results.append(result)
  page.close()
 b.close()
(OUT/'results.json').write_text(json.dumps({'results':results,'pageErrors':errors},ensure_ascii=False,indent=2)+'\n')
print(json.dumps({'page_viewport_checks':len(results),'viewports':[1280,390],'pages':len(pages),'images_loaded':sum(len(r['images']) for r in results),'pageErrors':errors},ensure_ascii=False))
